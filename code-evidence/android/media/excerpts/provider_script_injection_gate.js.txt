// Load the HHTQ intervention only inside its player frame. Keep the implementation out of
// the script served to unrelated sites, and use the existing local JavaScript asset route.
(function ibLoadHhtqIntervention() {
    if (location.origin !== "https://streamfree.vip" ||
            !/^\/embed\/(v|vt)\/[a-zA-Z0-9_-]+$/.test(location.pathname) ||
            window === window.top || window.ibHhtqScriptLoaded) {
        return;
    }

    try {
        var parent = new URL(document.referrer);
        if (parent.protocol !== "https:" || !/^(www\.)?hhtq\.[a-z0-9-]+$/.test(parent.hostname)) {
            return;
        }

        var source = document.currentScript;
        if (!source || !source.src) {
            return;
        }
        var script = document.createElement("script");
        script.src = new URL("specialsites/hhtq.js", source.src).href;
        script.async = false;
        window.ibHhtqScriptLoaded = true;
        (document.head || document.documentElement).appendChild(script);
    } catch (_) {
        // Invalid or missing frame metadata must not enable a site intervention.
    }
})();
